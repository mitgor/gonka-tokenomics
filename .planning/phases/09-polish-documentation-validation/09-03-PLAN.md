**Validate programmatically:**

After generation, run a validation script that checks all 5 workbooks:

```python
import os
from openpyxl import load_workbook

files = [
    "output/gonka_master_model.xlsx",
    "output/gonka_token_price.xlsx",
    "output/gonka_fee_transition.xlsx",
    "output/gonka_host_profitability.xlsx",
    "output/gonka_treasury_pol.xlsx",
]

for f in files:
    assert os.path.exists(f), f"Missing: {f}"
    size_mb = os.path.getsize(f) / (1024 * 1024)
    assert size_mb < 5, f"{f} too large: {size_mb:.1f}MB"

    wb = load_workbook(f, data_only=False)
    for ws in wb.worksheets:
        # Check protection
        assert ws.protection.sheet is True, f"{f} / {ws.title}: protection not enabled"
        # Check print area exists
        assert ws.print_area is not None and ws.print_area != '', f"{f} / {ws.title}: no print area"

    # Check confidence column on Assumptions tab
    ws_a = wb["Assumptions"]
    confidence_found = False
    for row in ws_a.iter_rows(min_col=5, max_col=5, min_row=4, max_row=200):
        for cell in row:
            if cell.value in ("HIGH", "MED", "LOW"):
                confidence_found = True
                break
        if confidence_found:
            break
    assert confidence_found, f"{f}: No confidence values in column E"

    print(f"PASS: {f} ({size_mb:.2f}MB, {len(wb.sheetnames)} tabs, all protected, all print areas set)")

print("\nAll 5 workbooks validated successfully.")
```
