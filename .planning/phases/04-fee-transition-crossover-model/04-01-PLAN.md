```python
def _add_tail_emission_toggle(ws, param_refs):
    """Add ON/OFF DataValidation to the Tail Emission Toggle parameter cell.

    The cell already exists from the PARAM_GROUPS loop. This function
    adds a dropdown constraint so the user can only enter ON or OFF.

    Args:
        ws: The Assumptions worksheet.
        param_refs: Dict containing "Tail Emission Toggle" -> "Assumptions!$B$NN".
    """
    # Parse row number from param_refs (format: "Assumptions!$B$NN")
    ref = param_refs["Tail Emission Toggle"]
    row = int(ref.split("$")[-1])

    toggle_cell = ws.cell(row=row, column=2)

    dv = DataValidation(
        type="list",
        formula1='"ON,OFF"',
        allow_blank=False,
    )
    dv.prompt = "Select ON or OFF"
    dv.promptTitle = "Tail Emission"
    ws.add_data_validation(dv)
    dv.add(toggle_cell)
```

```python
    # emission_value = IF(toggle, MAX(mining, tail_rate), mining) * price
    # We need fee_revenue_per_period > emission_value_per_period
    # Fee per period = devs*(1+g)^(epoch/365)*rpd * (days/365) but days varies
    # Emission per period = mining_GNK * price (mining_GNK is already per-period)
    #
    # For annual comparison (avoiding days array), compare annualized rates:
    # Annual fee = devs*(1+g)^(epoch/365)*rpd  (no days factor)
    # Annual emission value = mining_GNK/days*365 * price  (annualize the per-period emission)
    # But mining_GNK/days is complex as an array...
    #
    # SIMPLEST: just divide fee_column by emission_value and check >= 1
    # But fee columns C/D/E already have days/365, and G already has the period emission value
    # So the ratio in H/I/J IS correct per-period.
    # For matrix with different prices: ratio = fee_col / (effective_emission_GNK * matrix_price)
    # effective_emission_GNK = IF(toggle, MAX(mining, tail*days), mining)
    # But days varies per row...
    #
    # Resolution: use the SAME formula as columns C/G but with substituted price/growth.
    # Express days as: IF(ROW()-2<=26, 30, 365) which maps row 3-26 to 30 and row 27-34 to 365
    # ROW(A3)-2 = 1, ROW(A26)-2 = 24, ROW(A27)-2 = 25
    # Monthly rows: row 3-26 (i=0-23), days=30 → ROW()-2 <= 24 → IF(ROW(ref)<=26, 30, 365)

    days_expr = "IF(ROW(INDIRECT(\"A\"&ROW()))<=26,30,365)"

    fee = (
        f"{base_devs_ref}*(1+{growth_ref})^"
        f"({es_sheet}!{es_start_col}$3:{es_start_col}$34/365)"
        f"*{rev_per_dev_ref}*{days_expr}/365"
    )

    eff_emission = (
        f"IF({tail_toggle_ref}=\"ON\","
        f"MAX({es_sheet}!{es_mining_col}$3:{es_mining_col}$34,"
        f"{tail_rate_ref}*{days_expr}),"
        f"{es_sheet}!{es_mining_col}$3:{es_mining_col}$34)"
        f"*{tp_sheet}!{price_col}$3:{price_col}$34"
    )

    return (
        f"=IFERROR(INDEX($A$3:$A$34,"
        f"MATCH(TRUE,INDEX(({fee})/({eff_emission})>=1,),0)),"
        f"\"Never\")"
    )
```

## Anti-Patterns to Avoid

- **Do NOT hardcode parameter values** -- every value must come from param_refs or cross-sheet references
- **Do NOT use named ranges** -- use direct cell references (project convention from FEATURES.md)
- **Do NOT forget the days factor** -- multiply fee revenue by days/365 for each period (30/365 for monthly, 365/365 for annual)
- **Do NOT use CSE array formulas** -- the INDEX/MATCH(TRUE,INDEX(...)) pattern works without Ctrl+Shift+Enter
- **Do NOT create a stacked bar without overlap=100** -- that's Plan 02's job, but note for awareness
- **Do NOT apply conditional formatting** -- that's Plan 02's scope (charts + formatting)
- **Do NOT forget IFERROR wrappers** on division formulas (division by zero when emission value is 0)
